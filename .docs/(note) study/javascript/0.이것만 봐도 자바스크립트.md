변수를 만드는 형태는 다음과 같다
var 변수명 = 값;
let 변수명 = 값;
const 변수명 = 값;

각 특징은 다음과 같다.
- var
```javascript
// 중복 선언시에는 키워드가 무시된다. = 중복 선언 가능!
var a = 1;
var b = 1;
console.log(a); //1
console.log(b); //1
var a = 100; // 재할당만 한다.
var b; // 무시된다.
console.log(a); //100
console.log(b); //1

```
이유는? 변수 선언문을 먼저 다 위로 가져가서 처리해버리고 초기화문은 한 줄씩 읽으면서 처리하기 때문에 가능하다. (이 개념을 호이스팅Hoisting이라고 한다.)   

- var의 스코프
```javascript
//var는 오직 함수 코드블록만을 지역스코프로 인정한다.
var a = 1;

if (true) {
var a = 10;
} // var를 통해서 변수를 하나 더 만들어 처리하고 싶었다는 의도는 보이지만 
// 함수 아니죠? 바로 덮어씌우기.

console.log(a); // 10
```
- let
```javascript
//중복 선언이 금지된다.
let a = 1;
let a = 2; //SyntaxError: 식별자 'a'는 이미 선언되었습니다.
```
- let의 스코프
```javascript
// 모든 코드 블록{}을 지역스크프로 인정하는 블록레벨 스코프를 따른다.
let a = 1;
{
    let a = 2; // 이 코드 블럭 내에서 a라는 변수를 새롭게 만들어 바깥 a와 구분됨. 
    let b = 3; 

    console.log(a); // 2
    console.log(b); // 3
}// 참고로 지역변수는 해당 블록이 끝나면 제거된다.
console.log(a); // 1
console.log(b); // ReferenceError: b가 선언되지 않았습니다.
```
- let은 변수 호이스팅이 발생하지 않는 것처럼 동작한다.
> var의 경우에는 호이스팅으로 인해 변수가 한 번 선언되고 그 변수에 undefined 값을 넣어둔다.  
> let의 경우에는 var처럼 호이스팅으로 인해 선언문을 끌고가는데 undefined를 넣는 초기화 단계가 없다.

그럼 정말 let은 호이스팅이 발생하지 않는가?
```javascript
let a = 1;
{
    console.log(a); // referenceError: 'a'를 초기화 하기 전에는 접근할 수 없어요.
    let a = 2; //지역변수
}
```
만약 호이스팅이 발생하지 않는다면 위 코드블럭에서 a값은 전역변수 값을 가져와서 1을 출력해야 했다.  
하지만, 코드 블럭 내에서 호이스팅이 있었기 때문에, 초기화를 안 했다고 뜨는 것이다.

- 브라우저 환경에서 전역객체(window)와 let
```javascript
// 브라우저 환경에서 var나 function으로 생성된 전역 변수나 전역 함수는 전역객체(window)의 프로퍼티가 된다.
var x = 1;
y = 2; // 브라우저 환경에서 선언문 없이 이렇게 박아버린 경우에는 암묵적 전역 변수라 해서 window의 프로퍼티가 된다.
function z() {}

console.log(window.x); // 1
console.log(x); // 1
console.log(window.y); //2
console.log(y); //2
console.log(window.z); // function z() {}
console.log(z); // function z() {}

// let과 cosnt로 선언한 전역 변수는 전역 객체(window)의 프로퍼티가 아니다.
let a = 4;
console.log(window.a); // undefined
console.log(a); // 4
```
- const  
상수를 선언하기 위해 선언되는데, let이랑 대부분 동일한 특징을 가지기 때문에 let과 다른 점만 보겠다.
- const의 선언과 초기화
> const 키워드로 선언한 변수는 반드시 선언과 동시에 초기화 해야한다.
- 재할당이 금지된다.
- 상수로서 이용할 수 있다.
- 일반적으로 대문자로 선언해서 상수임을 명확히 나타내고 스네이크 케이스로 표현한다.
- const에 객체를 할당한 경우, const는 영원히 객체다. 단, 객체 내부의 값은 변경될 수 있다.

결론
1. 그냥 다 const를 사용하고
2. 재할당할 일이 있으면 let으로 바꾸고
3. var는 이번 프로젝트에서 쓰지 않는다. 함수가 아닌 경우에 코드블록으로 인정해주지 않기 때문에 java를 중점으로 배워온 우리는 사용시 var의 스코프를 헷갈려서 오류가 발생할 수 있다.


변수명 자리는 식별자 이름 규칙을 지켜야 한다.

### 식별자 네이밍 규칙

- 식별자는 특수문자를 제외한 문자, 숫자, 언더스코어, 달러 기호를 포함할 수 있다.
- 단, 식별자는 숫자로 시작할 수 없다.
- 예약어는 식별자로 사용할 수 없다.

- 변수는 쉼표를 사용해 한 번에 선언할 수 있다.
- 유니코드 문자로 명명된 식별자는 사용할 수 있지만 바람직하지 않다.
- 대소문자를 구별한다.
- 변수의 존재 목적을 쉽게 이해할 수 있도록 의미를 명확히 표현해야 한다.

일반적으로 변수,함수 = 카멜  
생성자, 클래스 = 파스칼을 사용한다.
